Problem Restatement

You are given a string s of lowercase English letters.

	
The length of s is a perfect square
	
Let n = √len(s)
	
Fill an n × n grid row-wise (left to right, top to bottom)
	
Count:
	
 
Rows where all characters are the same
 
Columns where all characters are the same
	
	
	
Return the total count (rows + columns)

Approach

Step 1: Compute grid size

n = int(sqrt(len(s)))

Step 2: Build the grid

Fill the grid row by row:

grid[i][j] = s[i*n + j]

Step 3: Count uniform rows

A row is valid if:

row[0] == row[1] == ... == row[n-1]

Step 4: Count uniform columns

A column is valid if:

grid[0][j] == grid[1][j] == ... == grid[n-1][j]

Python Implementation

import math

 

def count_uniform_rows_and_columns(s):

 n = int(math.sqrt(len(s)))

 

 # Build the grid

 grid = []

 idx = 0

 for i in range(n):

 grid.append(s[idx:idx+n])

 idx += n

 

 count = 0

 

 # Count uniform rows

 for row in grid:

 if all(ch == row[0] for ch in row):

 count += 1

 

 # Count uniform columns

 for col in range(n):

 first_char = grid[0][col]

 if all(grid[row][col] == first_char for row in range(n)):

 count += 1

 

 return count

Java Solution

public class UniformGridCount {

 

 public static int countUniformRowsAndColumns(String s) {

 int len = s.length();

 int n = (int) Math.sqrt(len);

 

 // Build the grid

 char[][] grid = new char[n][n];

 int index = 0;

 

 for (int i = 0; i < n; i++) {

 for (int j = 0; j < n; j++) {

 grid[i][j] = s.charAt(index++);

 }

 }

 

 int count = 0;

 

 // Check uniform rows

 for (int i = 0; i < n; i++) {

 boolean isUniform = true;

 for (int j = 1; j < n; j++) {

 if (grid[i][j] != grid[i][0]) {

 isUniform = false;

 break;

 }

 }

 if (isUniform) {

 count++;

 }

 }

 

 // Check uniform columns

 for (int j = 0; j < n; j++) {

 boolean isUniform = true;

 for (int i = 1; i < n; i++) {

 if (grid[i][j] != grid[0][j]) {

 isUniform = false;

 break;

 }

 }

 if (isUniform) {

 count++;

 }

 }

 

 return count;

 }

 

 public static void main(String[] args) {

 String s = "aaaabbbcc";

 System.out.println(countUniformRowsAndColumns(s));

 }

}
